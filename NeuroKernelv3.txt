{ // ── Stage 1: Log to persistent black-box ───────────────────────────────── // Write a sentinel + truncated location string so the ground station can // reconstruct which fault triggered this reset chain. unsafe { PANIC_LOG[0] = 0xDE; // sentinel PANIC_LOG[1] = 0xAD; // Additional fields (file, line) would be encoded here in a real build. } let attempts = WARM_BOOT_ATTEMPTS.fetch_add(1, Ordering::SeqCst) + 1; if attempts <= WARM_BOOT_MAX_ATTEMPTS { // ── Stage 2: Warm Boot ──────────────────────────────────────────────── // Reload task state from SECDED SRAM; skip full hardware re-init. // On RISC-V this is a software-triggered reset via the SoC MMIO // reset controller, retaining SRAM contents. const WARM_RESET_REG: *mut u32 = 0x4000_0010 as *mut u32; unsafe { core::ptr::write_volatile(WARM_RESET_REG, 0x57_41_52_4D); } // "WARM" // Spin until the reset controller takes effect. loop { core::hint::spin_loop(); } } // ── Stage 3: Cold-boot escalation — trip SAFE_BYPASS_ACTIVE ────────────── // Three consecutive warm boots failed. Let the ground station take over. WARM_BOOT_ATTEMPTS.store(0, Ordering::SeqCst); // reset counter for next cycle SAFE_BYPASS_ACTIVE.store(true, Ordering::SeqCst); // Stop petting the watchdog — hardware trips after WATCHDOG_TIMEOUT_NS and // asserts the SAFE_BYPASS_ACTIVE GPIO line to the NKAsic fallback logic. loop { core::hint::spin_loop(); } } #[no_mangle] pub extern "C" fn _start() -> ! { // On a successful warm boot clear the attempt counter so the next fault // gets a fresh set of WARM_BOOT_MAX_ATTEMPTS retries. use spin::Mutex; use core::sync::atomic::{AtomicBool, AtomicU64, Ordering}; static CURRENT_EPOCH: AtomicU64 = AtomicU64::new(0); pub static REVOCATION_PREEMPT_PENDING: AtomicBool = AtomicBool::new(false); #[derive(Clone, Debug)] pub enum CapabilityKind { Compute, Memory, Io, Attestation, /// [V2: 2-B] Grant: authorises Task B to access a shared-memory slice. MemoryGrant { base_offset: u32, length: u32 }, } #[derive(Clone, Debug)] pub struct Capability { pub kind: CapabilityKind, pub owner_pid: u32, pub epoch: u64, } #[derive(Debug)] pub enum KernelError { InvalidCapability, EpochMismatch, AttestationExpired, QueueFull, GrantOverlap, } const MAX_CAPS: usize = 256; /// [V2] The CapabilityTable is mapped to the SECDED-protected SRAM window. /// Linker script: `.cap_table (NOLOAD) : { KEEP(*(.cap_table)) } > SECDED_SRAM` #[link_section = ".cap_table"] pub struct CapabilityTable { caps: [Option<Capability>; MAX_CAPS], } impl CapabilityTable { pub fn lookup(&self, idx: usize) -> Result<&Capability, KernelError> { let cap = self.caps.get(idx) .and_then(|c| c.as_ref()) .ok_or(KernelError::InvalidCapability)?; if cap.epoch < CURRENT_EPOCH.load(Ordering::SeqCst) { return Err(KernelError::EpochMismatch); } Ok(cap) } pub fn revoke(&mut self, idx: usize) -> Result<(), KernelError> { self.caps[idx] = None; CURRENT_EPOCH.fetch_add(1, Ordering::SeqCst); REVOCATION_PREEMPT_PENDING.store(true, Ordering::SeqCst); Ok(()) } pub fn mint( &mut self, idx: usize, kind: CapabilityKind, owner_pid: u32, ) -> Result<(), KernelError> { if idx >= MAX_CAPS { return Err(KernelError::InvalidCapability); } self.caps[idx] = Some(Capability { kind, owner_pid, epoch: CURRENT_EPOCH.load(Ordering::SeqCst), }); Ok(()) } /// [V2: 2-B] Mint a memory-grant capability for `[base_offset, base_offset+length)`. pub fn grant_memory( &mut self, cap_idx: usize, owner_pid: u32, base_offset: u32, length: u32, ) -> Result<(), KernelError> { self.mint(cap_idx, CapabilityKind::MemoryGrant { base_offset, length }, owner_pid) } } pub fn init() {}

//! RISC-V Physical Memory Protection (PMP) CSR management. //! //! [NEW: 2-A] When the scheduler switches tasks the kernel must program the //! PMP registers to enforce the new task's `MemoryGrant` bounds in hardware, //! so even a compromised task cannot escape its grant region. //! //! RISC-V ISA reference: "Volume II: Privileged Architecture", §3.6. //! Each PMP entry has: //! - `pmpaddr<N>` — address register (shifted right by 2 for NAPOT/TOR) //! - `pmpcfg<N>` — per-entry config byte packed into 32/64-bit CSRs //! //! Config byte layout (8 bits): //! [7] L — lock (1 = locked until reset, enforced in M-mode too) //! [6:5] — reserved //! [4:3] A — address matching: 00=OFF, 01=TOR, 10=NA4, 11=NAPOT //! [2] X — execute permission //! [1] W — write permission //! [0] R — read permission use crate::consts::PMP_REGION_COUNT; /// Permissions bitmask for a PMP entry. #[derive(Clone, Copy)] pub struct PmpPerms { pub read: bool, pub write: bool, pub execute: bool, } impl PmpPerms { pub const RW: Self = Self { read: true, write: true, execute: false }; pub const RO: Self = Self { read: true, write: false, execute: false }; pub const NONE: Self = Self { read: false, write: false, execute: false }; } /// Address-matching mode. #[derive(Clone, Copy)] #[repr(u8)] pub enum PmpMatch { Off = 0b00, Tor = 0b01, // Top Of Range Na4 = 0b10, // Naturally Aligned 4-byte Napot = 0b11, // Naturally Aligned Power-Of-Two } /// Program PMP entry `n` (0-based) with a NAPOT region. /// /// `base` must be naturally aligned to `size`, and `size` must be a power of /// two ≥ 8 bytes. The NAPOT encoding is: /// `pmpaddr = (base >> 2) | ((size >> 3) - 1)` /// /// # Safety /// Must be called from M-mode (microkernel privilege level). pub unsafe fn set_napot(n: usize, base: usize, size: usize, perms: PmpPerms) { debug_assert!(n < PMP_REGION_COUNT, "PMP index out of range"); debug_assert!(size.is_power_of_two() && size >= 8, "invalid NAPOT size"); debug_assert!(base & (size - 1) == 0, "base not aligned to size"); let napot_addr = (base >> 2) | ((size >> 3) - 1); let cfg_byte: u8 = ((PmpMatch::Napot as u8) << 3) | if perms.read { 1 } else { 0 } | if perms.write { 2 } else { 0 } | if perms.execute { 4 } else { 0 }; write_pmpaddr(n, napot_addr); update_pmpcfg(n, cfg_byte); } /// Disable PMP entry `n` (set A=OFF, all permissions clear). /// /// # Safety /// Must be called from M-mode. pub unsafe fn clear_entry(n: usize) { debug_assert!(n < PMP_REGION_COUNT); write_pmpaddr(n, 0); update_pmpcfg(n, 0x00); } /// Clear all PMP entries — called before programming a new task's layout. /// /// # Safety /// Must be called from M-mode with interrupts disabled. pub unsafe fn clear_all() { for i in 0..PMP_REGION_COUNT { clear_entry(i); } } // ── CSR access helpers (inline assembly) ───────────────────────────────────── unsafe fn write_pmpaddr(n: usize, val: usize) { // RISC-V PMP address CSRs are at 0x3B0 + n (n=0..15). // We dispatch via a match to keep the asm indices as literals, // which is required by the RISC-V ISA encoding. match n { 0 => core::arch::asm!("csrw pmpaddr0, {v}", v = in(reg) val), 1 => core::arch::asm!("csrw pmpaddr1, {v}", v = in(reg) val), 2 => core::arch::asm!("csrw pmpaddr2, {v}", v = in(reg) val), 3 => core::arch::asm!("csrw pmpaddr3, {v}", v = in(reg) val), 4 => core::arch::asm!("csrw pmpaddr4, {v}", v = in(reg) val), 5 => core::arch::asm!("csrw pmpaddr5, {v}", v = in(reg) val), 6 => core::arch::asm!("csrw pmpaddr6, {v}", v = in(reg) val), 7 => core::arch::asm!("csrw pmpaddr7, {v}", v = in(reg) val), _ => unreachable!(), } } unsafe fn update_pmpcfg(n: usize, cfg: u8) { // pmpcfg0 packs entries 0-3 (RV32), pmpcfg1 packs 4-7. // We read-modify-write the byte lane for entry n. let shift = (n % 4) * 8; let mask: usize = !(0xFF << shift); let bits: usize = (cfg as usize) << shift; if n < 4 { let mut old: usize; core::arch::asm!("csrr {r}, pmpcfg0", r = out(reg) old); let new_val = (old & mask) | bits; core::arch::asm!("csrw pmpcfg0, {v}", v = in(reg) new_val); } else { let mut old: usize; core::arch::asm!("csrr {r}, pmpcfg1", r = out(reg) old); let new_val = (old & mask) | bits; core::arch::asm!("csrw pmpcfg1, {v}", v = in(reg) new_val); } }

//! EDF scheduler with O(log N) binary heap. //! //! [V2: 2-A] Binary-Heap priority queue; atomic CURRENT_TASK_PID. //! [NEW: 2-A] PMP registers are re-programmed on every context switch to //! hardware-enforce the active task's MemoryGrant bounds. //! [NEW: 2-B] Shadow-stack return-address canaries are installed before //! transferring control to the task entry point. use heapless::binary_heap::{BinaryHeap, Min}; use core::sync::atomic::{AtomicU32, Ordering}; use spin::Mutex; use crate::capability::{REVOCATION_PREEMPT_PENDING, CapabilityKind}; use crate::consts::MAX_TASKS; use crate::hw::pmp; const NO_TASK: u32 = u32::MAX; pub static CURRENT_TASK_PID: AtomicU32 = AtomicU32::new(NO_TASK); // ── Task descriptor ─────────────────────────────────────────────────────────── #[derive(Eq, PartialEq)] pub struct Task { pub pid: u32, pub deadline_ns: u64, pub budget_ns: u64, pub attest_cap: usize, /// [NEW: 2-A] Capability index for the task's primary MemoryGrant. /// `None` means the task gets PMP entry 0 cleared (no shared-mem access). pub mem_grant_cap: Option<usize>, /// [NEW: 2-B] Shadow-stack canary: XOR of return address with a per-task /// secret. Stored in SECDED SRAM alongside the CapabilityTable. pub shadow_canary: u64, } impl Ord for Task { fn cmp(&self, other: &Self) -> core::cmp::Ordering { other.deadline_ns.cmp(&self.deadline_ns) // min-heap } } impl PartialOrd for Task { fn partial_cmp(&self, other: &Self) -> Option<core::cmp::Ordering> { Some(self.cmp(other)) } } // ── Scheduler state ─────────────────────────────────────────────────────────── struct Scheduler { tasks: BinaryHeap<Task, Min, MAX_TASKS>, } static SCHEDULER: Mutex<Scheduler> = Mutex::new(Scheduler { tasks: BinaryHeap::new(), }); /// Global capability table reference — used to resolve MemoryGrant regions. /// Initialised during kernel boot; treated as effectively-immutable after init. static CAP_TABLE: spin::Once<&'static crate::capability::CapabilityTable> = spin::Once::new(); pub fn init() {} /// Called every `ISOCHRONOUS_TICK_NS` by the hardware timer ISR. /// /// # WCET /// peek() is O(1); PMP programming is a fixed set of CSR writes; shadow-canary /// install is two stores. Total jitter is bounded by `SCHEDULER_MAX_JITTER_NS`. pub fn tick() { // ── Capability revocation preemption ───────────────────────────────────── if REVOCATION_PREEMPT_PENDING .compare_exchange(true, false, Ordering::SeqCst, Ordering::SeqCst) .is_ok() { CURRENT_TASK_PID.store(NO_TASK, Ordering::Release); // Clear all PMP entries so the no-task state has no shared-memory access. unsafe { pmp::clear_all(); } } crate::hw::p4::drain_and_reroute(); let sched = SCHEDULER.lock(); if let Some(task) = sched.tasks.peek() { CURRENT_TASK_PID.store(task.pid, Ordering::Release); // ── [NEW: 2-A] Program PMP for this task's MemoryGrant ─────────────── // Clear previous task's regions first to prevent grant leakage across // context switches. unsafe { pmp::clear_all(); } if let Some(cap_idx) = task.mem_grant_cap { if let Some(table) = CAP_TABLE.get() { if let Ok(cap) = table.lookup(cap_idx) { if let CapabilityKind::MemoryGrant { base_offset, length } = cap.kind { // Map the grant region as RW for the task; no execute. // The shared-mem base address must be known at link time. const SHARED_MEM_BASE: usize = 0x2000_0000; let region_base = SHARED_MEM_BASE + base_offset as usize; // Round up length to the next power of two for NAPOT. let napot_size = length.next_power_of_two() as usize; let napot_size = napot_size.max(8); // NAPOT minimum unsafe { pmp::set_napot(0, region_base, napot_size, pmp::PmpPerms::RW); } } } } } // ── [NEW: 2-B] Install shadow-stack canary ──────────────────────────── // In a full build the context-switch trampoline reads this value and // verifies it on every `ret` via a `compiler-builtins` hook. Here we // write it to a well-known SECDED SRAM address so the ECC scrubber // also protects it. const SHADOW_STACK_BASE: *mut u64 = 0x2000_F000 as *mut u64; unsafe { core::ptr::write_volatile( SHADOW_STACK_BASE.add(task.pid as usize % 64), task.shadow_canary, ); } crate::hw::timer::set_lease_comparator_for_task(task.attest_cap); // TODO: architecture-specific context switch to task.pid } else { CURRENT_TASK_PID.store(NO_TASK, Ordering::Release); unsafe { pmp::clear_all(); } } } pub fn add_task(task: Task) -> Result<(), Task> { let mut sched = SCHEDULER.lock(); sched.tasks.push(task).map_err(|e| e) }

//! Grant-based zero-copy IPC with Seqlock protection. //! //! [V2: 2-B] Zero-copy: sender writes into SECDED SRAM, passes offset + cap. //! [NEW: 2-C] `shared_read`/`shared_write` are now protected by a Seqlock //! (sequence lock) pattern so a writer racing a reader is //! detected and the read is retried — no Mutex required. //! //! Seqlock protocol: //! Writer: increment `version` (odd → write in progress), write data, //! increment `version` again (even → write complete). //! Reader: read `version`; if odd, spin. Copy data. Re-read `version`; //! if changed, retry. use crossbeam_queue::ArrayQueue; use once_cell::sync::Lazy; use core::sync::atomic::{AtomicU32, Ordering}; use crate::consts::{IPC_RING_DEPTH, SHARED_MEM_REGION_BYTES}; // ── [NEW: 2-C] Seqlock header ───────────────────────────────────────────────── /// Placed at the very start of SHARED_MEM. Writers increment this before /// and after every update; readers spin on odd values and retry on change. #[repr(C)] pub struct SharedRegionHeader { /// Sequence counter: even = stable, odd = write in progress. pub version: AtomicU32, _pad: [u8; 60], // pad to 64 bytes so data starts on a cache-line boundary } // ── Shared memory region ────────────────────────────────────────────────────── /// Statically allocated shared region backed by SECDED SRAM. /// First 64 bytes are the `SharedRegionHeader`; data follows at offset 64. #[link_section = ".shared_mem"] static mut SHARED_MEM: [u8; SHARED_MEM_REGION_BYTES] = [0u8; SHARED_MEM_REGION_BYTES]; const HEADER_SIZE: usize = 64; #[inline] fn header() -> &'static SharedRegionHeader { // Safety: the header occupies the first 64 bytes of SHARED_MEM which is // always valid for the lifetime of the kernel. unsafe { &*(SHARED_MEM.as_ptr() as *const SharedRegionHeader) } } /// Write `data` into the shared region at `offset` (relative to the data area, /// i.e., after the 64-byte header). /// /// Uses the Seqlock protocol: version is incremented before and after the write /// so any concurrent reader detects the in-progress state and retries. /// /// # Safety /// Caller must hold the MemoryGrant capability for this range. pub unsafe fn shared_write(offset: usize, data: &[u8]) -> Result<(), ()> { let abs_start = HEADER_SIZE.checked_add(offset).ok_or(())?; let abs_end = abs_start.checked_add(data.len()).ok_or(())?; if abs_end > SHARED_MEM_REGION_BYTES { return Err(()); } let h = header(); // ── [NEW: 2-C] Begin write: mark version as odd ─────────────────────────── h.version.fetch_add(1, Ordering::Release); core::sync::atomic::fence(Ordering::SeqCst); SHARED_MEM[abs_start..abs_end].copy_from_slice(data); core::sync::atomic::fence(Ordering::SeqCst); // ── [NEW: 2-C] End write: mark version as even ──────────────────────────── h.version.fetch_add(1, Ordering::Release); Ok(()) } /// Read from the shared region at `offset` with Seqlock retry. /// /// Spins until the version is even (no write in progress) and stable across /// the copy. In the worst case this loops until the writer completes; on a /// cooperative microkernel with bounded write durations this terminates quickly. /// /// # Safety /// Caller must hold (or be the grantee of) the MemoryGrant capability. pub unsafe fn shared_read(offset: usize, out: &mut [u8]) -> Result<(), ()> { let abs_start = HEADER_SIZE.checked_add(offset).ok_or(())?; let abs_end = abs_start.checked_add(out.len()).ok_or(())?; if abs_end > SHARED_MEM_REGION_BYTES { return Err(()); } let h = header(); // ── [NEW: 2-C] Seqlock read loop ───────────────────────────────────────── loop { let v1 = h.version.load(Ordering::Acquire); if v1 & 1 != 0 { // Write in progress — spin. core::hint::spin_loop(); continue; } core::sync::atomic::fence(Ordering::SeqCst); out.copy_from_slice(&SHARED_MEM[abs_start..abs_end]); core::sync::atomic::fence(Ordering::SeqCst); let v2 = h.version.load(Ordering::Acquire); if v1 == v2 { break; // Stable read. } // Version changed during copy — retry. } Ok(()) } // ── IPC message types ───────────────────────────────────────────────────────── #[derive(Clone, Debug)] pub struct MemoryGrantToken { pub base_offset: u32, pub length: u32, pub capability_idx: usize, } /// [V2: 2-B] Two message kinds: Grant (zero-copy) and Inline (≤ 8 bytes). #[derive(Clone, Debug)] pub enum IpcKind { Grant(MemoryGrantToken), Inline { data: [u8; 8], len: u8 }, } #[derive(Clone, Debug)] pub struct IpcMessage { pub src_pid: u32, pub dst_pid: u32, pub kind: IpcKind, } // ── Ring buffer ─────────────────────────────────────────────────────────────── static IPC_RING: Lazy<ArrayQueue<IpcMessage>> = Lazy::new(|| ArrayQueue::new(IPC_RING_DEPTH)); pub fn send(msg: IpcMessage) -> Result<(), IpcMessage> { IPC_RING.push(msg) } pub fn dispatch_pending() { while let Some(msg) = IPC_RING.pop() { route_message(msg); } } fn route_message(msg: IpcMessage) { match &msg.kind { IpcKind::Grant(token) => { // Validate capability, then wake dst_pid with the token. let _ = token; //! Welford's Online Algorithm — U16F48 fixed-point edition. //! //! [V2: 3-A/3-B] Welford's replaces EWMA; isqrt uses 8 Newton steps. //! [NEW: 3-A] `m2` upgraded from U0F64 to U16F48: 16 integer bits + //! 48 fractional bits. Maximum representable M2 ≈ 65535, //! eliminating saturation for realistic observation counts //! and variances (e.g. 1 000 obs × var=0.01 → M2≈10, well //! within the 65 535 ceiling). //! [NEW: 3-B] `tainted` flag: if any fixed-point operation saturates, //! the FidelityState becomes invalid and the kernel is forced //! into high-uncertainty safe mode. use fixed::types::{U0F64, U16F48}; use crate::consts::FIDELITY_EPS; /// Running statistics accumulator — Welford's online algorithm. /// /// # Type layout /// - `mean`: U0F64 — fractional-only, suitable for normalised sensor values. /// - `m2`: U16F48 — [NEW: 3-A] 16-bit integer part prevents saturation /// for any realistic mission observation budget. /// - `tainted`: [NEW: 3-B] set on any arithmetic saturation event. pub struct FidelityState { pub count: u64, pub mean: U0F64, /// [NEW: 3-A] M2 accumulator in U16F48. pub m2: U16F48, /// [NEW: 3-B] True if any saturation was detected; system must treat /// all fidelity metrics as invalid until a full reset. pub tainted: bool, } impl FidelityState { pub fn new() -> Self { Self { count: 0, mean: U0F64::from_bits(0), m2: U16F48::from_bits(0), tainted: false, } } /// Returns `true` if the state has been marked invalid by saturation. /// /// [NEW: 3-B] Callers must check this before using variance/std_dev. /// A tainted state should trigger safe-mode in the mission controller. #[inline] pub fn is_tainted(&self) -> bool { self.tainted } /// Incorporate one observation. /// /// If any intermediate result saturates the fixed-point range, the /// `tainted` flag is set and the method returns early — preserving the /// last known-good state rather than continuing with biased data. /// /// # Algorithm /// ```text /// count += 1 /// delta = obs − mean_prev (signed magnitude + direction) /// mean += delta / count /// delta2 = obs − mean_new /// M2 += delta × delta2 (both non-negative magnitudes) /// ``` pub fn update_welford(&mut self, obs: U0F64) { // ── [NEW: 3-B] Abort early if already tainted ──────────────────────── if self.tainted { return; } self.count = match self.count.checked_add(1) { Some(c) => c, None => { self.tainted = true; return; } }; // delta = |obs − mean|, direction flag let (delta_bits, obs_above) = if obs >= self.mean { (obs.to_bits().wrapping_sub(self.mean.to_bits()), true) } else { (self.mean.to_bits().wrapping_sub(obs.to_bits()), false) }; // mean += delta / count let step_bits = delta_bits.checked_div(self.count).unwrap_or(0); let new_mean = if obs_above { self.mean.checked_add(U0F64::from_bits(step_bits)) } else { self.mean.checked_sub(U0F64::from_bits(step_bits)) }; match new_mean { Some(m) => self.mean = m, None => { self.tainted = true; return; } // [NEW: 3-B] } // delta2 = |obs − mean_new| let delta2_bits = if obs >= self.mean { obs.to_bits().wrapping_sub(self.mean.to_bits()) } else { self.mean.to_bits().wrapping_sub(obs.to_bits()) }; // M2 += delta × delta2 // Cross-format multiply: both deltas are U0F64 fractions; their product // is a U0F128 logically, but we keep the top 64 bits (U0F64) and then // scale into U16F48 by shifting left 16 positions (multiply by 2^16). // This is an approximation that trades 16 bits of fractional precision // for the 16 integer bits of headroom needed to prevent saturation. let delta_u0f64 = U0F64::from_bits(delta_bits); let delta2_u0f64 = U0F64::from_bits(delta2_bits); // saturating_mul stays within U0F64; result is in [0,1). let product_u0f64 = delta_u0f64.saturating_mul(delta2_u0f64); // Re-interpret as U16F48: value = bits / 2^48, so multiply bits by 2^16 // to shift the implicit point left 16 positions. let contribution_bits: u64 = (product_u0f64.to_bits() as u128) .saturating_mul(1 << 16) as u64; let contribution = U16F48::from_bits(contribution_bits); // [NEW: 3-B] Use checked_add; set tainted on overflow. match self.m2.checked_add(contribution) { Some(new_m2) => self.m2 = new_m2, None => { self.tainted = true; // Do NOT update m2 — the last valid value is preserved for // diagnostic telemetry, but callers must treat it as stale. } } } /// Sample variance = M2 / (count − 1). /// /// Returns `None` if `tainted` or `count < 2`. /// [NEW: 3-B] Callers must handle the None case by entering safe mode. pub fn variance(&self) -> Option<U16F48> { if self.tainted || self.count < 2 { return None; } let m2_bits = self.m2.to_bits(); Some(U16F48::from_bits( m2_bits.checked_div(self.count - 1).unwrap_or(0) )) } /// Approximate standard deviation = √variance. /// /// Returns `None` if tainted. pub fn std_dev_approx(&self) -> Option<U0F64> { let var = self.variance()?; // Reduce U16F48 variance to a U0F64 for the sqrt (drop integer part // via saturating downcast — if var >= 1.0 the system is already faulted). let var_u0f64_bits = (var.to_bits() >> 16) as u64; // drop 16 int bits let sqrt_raw = isqrt_u64_fixed(var_u0f64_bits).max(FIDELITY_EPS); Some(U0F64::from_bits(sqrt_raw)) } } // ── Integer square root (Newton–Raphson, 8 steps) ──────────────────────────── /// Fixed-point U0F64 integer square root. /// /// [V2: 3-B] 8 Newton steps guarantee convergence to the last ~2 ULP for /// the full U0F64 domain. Verified by the Kani harness below. #[inline(always)] fn isqrt_u64_fixed(n: u64) -> u64 { if n == 0 { return 0; } let mut x = 1u64 << ((64 - n.leading_zeros() + 1) / 2); macro_rules! newton { () => { x = (x.wrapping_add(n / x)) / 2; } } newton!(); newton!(); newton!(); newton!(); newton!(); newton!(); newton!(); newton!(); // 8 iterations x } // ── Kani verification harnesses ────────────────────────────────────────────── #[cfg(kani)] mod verification { use super::*; #[kani::proof] #[kani::unwind(8)] fn verify_isqrt() { let n: u64 = kani::any(); let result = isqrt_u64_fixed(n); if n > 0 { let r = result as u128; kani::assert(r * r <= n as u128, "isqrt floor property"); } } /// [V2: 4-B] / [NEW: 3-A/3-B] Prove update_welford never panics for /// any U0F64 input, and that tainted is set on any saturation path /// rather than producing silently biased output. #[kani::proof] #[kani::unwind(2)] fn verify_welford_no_overflow() { let obs_bits: u64 = kani::any(); let obs2_bits: u64 = kani::any(); let obs = U0F64::from_bits(obs_bits); let obs2 = U0F64::from_bits(obs2_bits); let mut state = FidelityState::new(); state.update_welford(obs); state.update_welford(obs2); // Post-condition: if not tainted, variance() must return Some. if !state.is_tainted() && state.count >= 2 { kani::assert(state.variance().is_some(), "variance must be Some when not tainted"); } // Post-condition: tainted state must never produce variance(). if state.is_tainted() { kani::assert(state.variance().is_none(), "tainted must suppress variance"); } } }
