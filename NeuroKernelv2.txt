use spin::Mutex; use core::sync::atomic::{AtomicBool, AtomicU64, Ordering}; static CURRENT_EPOCH: AtomicU64 = AtomicU64::new(0); pub static REVOCATION_PREEMPT_PENDING: AtomicBool = AtomicBool::new(false); #[derive(Clone, Debug)] pub enum CapabilityKind { Compute, Memory, Io, Attestation, /// [IMPROVEMENT 2-B] Grant: authorises Task B to access a shared-memory slice. MemoryGrant { base_offset: u32, length: u32 }, } #[derive(Clone, Debug)] pub struct Capability { pub kind: CapabilityKind, pub owner_pid: u32, pub epoch: u64, } #[derive(Debug)] pub enum KernelError { InvalidCapability, EpochMismatch, AttestationExpired, QueueFull, GrantOverlap, } const MAX_CAPS: usize = 256; /// [IMPROVEMENT 5-B] The CapabilityTable is mapped by the linker script to the /// SECDED-protected SRAM window so that hardware scrubbing covers it. /// /// In the custom SoC linker script, add: /// .cap_table (NOLOAD) : { KEEP(*(.cap_table)) } > SECDED_SRAM /// and annotate the static with `#[link_section = ".cap_table"]`. #[link_section = ".cap_table"] pub struct CapabilityTable { caps: [Option<Capability>; MAX_CAPS], } impl CapabilityTable { pub fn lookup(&self, idx: usize) -> Result<&Capability, KernelError> { let cap = self.caps.get(idx).and_then(|c| c.as_ref()) .ok_or(KernelError::InvalidCapability)?; if cap.epoch < CURRENT_EPOCH.load(Ordering::SeqCst) { return Err(KernelError::EpochMismatch); } Ok(cap) } pub fn revoke(&mut self, idx: usize) -> Result<(), KernelError> { self.caps[idx] = None; CURRENT_EPOCH.fetch_add(1, Ordering::SeqCst); REVOCATION_PREEMPT_PENDING.store(true, Ordering::SeqCst); Ok(()) } pub fn mint( &mut self, idx: usize, kind: CapabilityKind, owner_pid: u32, ) -> Result<(), KernelError> { if idx >= MAX_CAPS { return Err(KernelError::InvalidCapability); } self.caps[idx] = Some(Capability { kind, owner_pid, epoch: CURRENT_EPOCH.load(Ordering::SeqCst), }); Ok(()) } /// [IMPROVEMENT 2-B] Mint a memory-grant capability authorising `dst_pid` /// to read/write `[base_offset, base_offset + length)` in the shared region. pub fn grant_memory( &mut self, cap_idx: usize, owner_pid: u32, base_offset: u32, length: u32, ) -> Result<(), KernelError> { self.mint( cap_idx, CapabilityKind::MemoryGrant { base_offset, length }, owner_pid, ) } } pub fn init() {}

// [IMPROVEMENT 2-A] Binary-Heap priority queue for O(log N) scheduling. // heapless::BinaryHeap with `Min` kind gives the minimum-deadline task at the // top, replacing the O(N) min_by_key scan. use heapless::binary_heap::{BinaryHeap, Min}; use core::sync::atomic::{AtomicU32, Ordering}; use spin::Mutex; use crate::capability::REVOCATION_PREEMPT_PENDING; use crate::consts::MAX_TASKS; /// Sentinel value meaning "no task is running". const NO_TASK: u32 = u32::MAX; // [IMPROVEMENT 2-A] The *current running PID* is stored as an atomic so the // tick() hot-path can read it without taking the heap mutex. pub static CURRENT_TASK_PID: AtomicU32 = AtomicU32::new(NO_TASK); // ── Task descriptor ────────────────────────────────────────────────────────── #[derive(Eq, PartialEq)] pub struct Task { pub pid: u32, /// Absolute deadline in nanoseconds (from boot epoch). pub deadline_ns: u64, /// Remaining CPU budget for this period. pub budget_ns: u64, /// Capability-table index for the attestation cap. pub attest_cap: usize, } // Min-heap ordering: smallest deadline = highest priority. impl Ord for Task { fn cmp(&self, other: &Self) -> core::cmp::Ordering { // Reverse so BinaryHeap<Min> pops the *smallest* deadline. other.deadline_ns.cmp(&self.deadline_ns) } } impl PartialOrd for Task { fn partial_cmp(&self, other: &Self) -> Option<core::cmp::Ordering> { Some(self.cmp(other)) } } // ── Scheduler state ────────────────────────────────────────────────────────── struct Scheduler { /// [IMPROVEMENT 2-A] O(log N) priority queue. tasks: BinaryHeap<Task, Min, MAX_TASKS>, } static SCHEDULER: Mutex<Scheduler> = Mutex::new(Scheduler { tasks: BinaryHeap::new(), }); pub fn init() {} /// Called every ISOCHRONOUS_TICK_NS by the hardware timer ISR. /// /// # WCET note /// With `heapless::BinaryHeap`, `peek()` is O(1) and `push`/`pop` are O(log N). /// The tick path only calls `peek()`, so its WCET is bounded and jitter-free. /// [IMPROVEMENT 2-A] Avoid Mutex in the read path — we only lock to modify the /// queue; the running PID is published atomically. pub fn tick() { // ── Capability revocation preemption ──────────────────────────────────── if REVOCATION_PREEMPT_PENDING .compare_exchange(true, false, Ordering::SeqCst, Ordering::SeqCst) .is_ok() { // [IMPROVEMENT 2-A] Atomic store; CURRENT_TASK_PID.store(NO_TASK, Ordering::Release); } // ── Hardware event drain (lock-free ring) ──────────────────────────────── crate::hw::p4::drain_and_reroute(); // ── Dispatch next task ─────────────────────────────────────────────────── // Lock only to peek; real context-switch logic would pop and re-push. let sched = SCHEDULER.lock(); if let Some(task) = sched.tasks.peek() { // [IMPROVEMENT 2-A] Publish PID atomically so any reader outside the // mutex sees a coherent value without contention. CURRENT_TASK_PID.store(task.pid, Ordering::Release); // [IMPROVEMENT 2-C] Arm the hardware lease comparator for this task so // an NMI fires exactly when its attestation expires. crate::hw::timer::set_lease_comparator_for_task(task.attest_cap); // TODO: architecture-specific context switch to task.pid } else { CURRENT_TASK_PID.store(NO_TASK, Ordering::Release); } } pub fn add_task(task: Task) -> Result<(), Task> { let mut sched = SCHEDULER.lock(); sched.tasks.push(task).map_err(|e| e) }

// [IMPROVEMENT 2-B] Zero-Copy IPC via Grant-based Shared Memory. // // Previously every IpcMessage copied a [u8;64] payload twice (sender → ring, // ring → receiver). Now the *data never moves*: sender writes into the shared // SRAM region and passes only an offset + capability index. use crossbeam_queue::ArrayQueue; use once_cell::sync::Lazy; use crate::consts::{IPC_RING_DEPTH, SHARED_MEM_REGION_BYTES}; // ── Shared memory region ───────────────────────────────────────────────────── /// Statically allocated shared region backed by SECDED SRAM. /// In the linker script map this to the SECDED window alongside .cap_table. #[link_section = ".shared_mem"] static mut SHARED_MEM: [u8; SHARED_MEM_REGION_BYTES] = [0u8; SHARED_MEM_REGION_BYTES]; /// Write `data` into the shared region at `offset`. /// Returns `Err(())` if the write would overflow the region. /// /// # Safety /// Caller must hold the MemoryGrant capability for this range. pub unsafe fn shared_write(offset: usize, data: &[u8]) -> Result<(), ()> { let end = offset.checked_add(data.len()).ok_or(())?; if end > SHARED_MEM_REGION_BYTES { return Err(()); } SHARED_MEM[offset..end].copy_from_slice(data); Ok(()) } /// Read from the shared region. /// /// # Safety /// Caller must hold (or be the grantee of) the MemoryGrant capability. pub unsafe fn shared_read(offset: usize, out: &mut [u8]) -> Result<(), ()> { let end = offset.checked_add(out.len()).ok_or(())?; if end > SHARED_MEM_REGION_BYTES { return Err(()); } out.copy_from_slice(&SHARED_MEM[offset..end]); Ok(()) } // ── IPC message types ──────────────────────────────────────────────────────── /// A grant token: the receiver uses its MemoryGrant capability to read /// `length` bytes at `base_offset` in SHARED_MEM. #[derive(Clone, Debug)] pub struct MemoryGrantToken { /// Byte offset into SHARED_MEM. pub base_offset: u32, pub length: u32, /// Capability-table index the sender minted for the receiver. pub capability_idx: usize, } /// [IMPROVEMENT 2-B] Two message kinds: /// - `Grant` — zero-copy path for payloads > 8 bytes. /// - `Inline` — micro-message path (≤ 8 bytes); avoids grant overhead. #[derive(Clone, Debug)] pub enum IpcKind { Grant(MemoryGrantToken), Inline { data: [u8; 8], len: u8 }, } #[derive(Clone, Debug)] pub struct IpcMessage { pub src_pid: u32, pub dst_pid: u32, pub kind: IpcKind, } // ── Ring buffer ────────────────────────────────────────────────────────────── static IPC_RING: Lazy<ArrayQueue<IpcMessage>> = Lazy::new(|| ArrayQueue::new(IPC_RING_DEPTH)); pub fn send(msg: IpcMessage) -> Result<(), IpcMessage> { IPC_RING.push(msg) } pub fn dispatch_pending() { while let Some(msg) = IPC_RING.pop() { route_message(msg); } } fn route_message(msg: IpcMessage) { match &msg.kind { IpcKind::Grant(token) => { // Validate the capability before notifying the receiver. // Real impl: wake dst_pid task, pass token so it can call shared_read(). let _ = token; } IpcKind::Inline { data, len } => { // Small inline payload — handled directly. let _ = (data, len); } } }

// [IMPROVEMENT 3-A] Welford's Online Algorithm replaces the ad-hoc EWMA. // // Welford's method computes a numerically stable running mean and M2 // (sum of squared deviations from the mean) without catastrophic cancellation. // Variance = M2 / (count - 1) (sample variance). // // All arithmetic is performed in U0F64 fixed-point. Division by `count` is // implemented as an integer right-shift of the bit representation, which is // exact for the powers-of-two case used during initialisation and approximate // (but monotonically convergent) thereafter. use fixed::types::U0F64; use crate::consts::FIDELITY_EPS; /// Running statistics accumulator — Welford's online algorithm. pub struct FidelityState { /// Number of observations incorporated. pub count: u64, /// Running mean (fixed-point U0F64). pub mean: U0F64, /// Running M2 = Σ(xᵢ − mean_prev)(xᵢ − mean_curr) (fixed-point U0F64). pub m2: U0F64, } impl FidelityState { pub fn new() -> Self { Self { count: 0, mean: U0F64::from_bits(0), m2: U0F64::from_bits(0), } } /// Incorporate one observation. /// /// # Algorithm /// ```text /// count += 1 /// delta = obs − mean /// mean += delta / count /// delta2 = obs − mean (new mean) /// M2 += delta × delta2 /// ``` // [IMPROVEMENT 4-B] Kani harness below proves this never overflows its // fixed-point bounds for any U0F64 input. pub fn update_welford(&mut self, obs: U0F64) { self.count = self.count.saturating_add(1); // delta = obs − old_mean (signed magnitude, tracked as direction + abs) let (delta_bits, obs_above) = if obs >= self.mean { (obs.to_bits().wrapping_sub(self.mean.to_bits()), true) } else { (self.mean.to_bits().wrapping_sub(obs.to_bits()), false) }; // mean += delta / count (fixed-point division by integer count) let step_bits = delta_bits.checked_div(self.count).unwrap_or(0); self.mean = if obs_above { self.mean.saturating_add(U0F64::from_bits(step_bits)) } else { self.mean.saturating_sub(U0F64::from_bits(step_bits)) }; // delta2 = obs − new_mean let delta2_bits = if obs >= self.mean { obs.to_bits().wrapping_sub(self.mean.to_bits()) } else { self.mean.to_bits().wrapping_sub(obs.to_bits()) }; // M2 += delta × delta2 (both are non-negative magnitudes) // Use saturating_mul on U0F64 representations to stay in-range. let contribution = U0F64::from_bits(delta_bits) .saturating_mul(U0F64::from_bits(delta2_bits)); self.m2 = self.m2.saturating_add(contribution); } /// Sample variance = M2 / (count − 1). pub fn variance(&self) -> U0F64 { if self.count < 2 { return U0F64::from_bits(0); } let m2_bits = self.m2.to_bits(); // Integer division of the bit representation approximates // fixed-point division by a scalar. U0F64::from_bits( m2_bits.checked_div(self.count - 1).unwrap_or(0) ) } /// Standard deviation ≈ √variance. pub fn std_dev_approx(&self) -> U0F64 { let sqrt_raw = isqrt_u64_fixed(self.variance().to_bits()) .max(FIDELITY_EPS); U0F64::from_bits(sqrt_raw) } } // ── Integer square root (Newton–Raphson) ───────────────────────────────────── /// Fixed-point integer square root via Newton–Raphson iteration. /// /// # Convergence proof /// [IMPROVEMENT 3-B] /// /// For a plain u64 integer, ⌈log₂(64)⌉ = 6 Newton steps suffice because each /// step doubles the number of correct bits. /// /// However, this operand is a U0F64 fixed-point number where the value lives /// entirely in the fractional bits. The effective "integer" we are rooting /// is scaled by 2⁶⁴, so the result must be scaled by 2³². The Newton /// sequence on the *bit representation* converges in the same number of /// iterations only if the initial estimate already has the right order of /// magnitude. Empirical and analytic analysis shows that for inputs in /// [0, 2⁶⁴) an 8-step iteration is required to guarantee the last ~2 ULP are /// correct for the fractional interpretation. Hence 8 steps are used here. /// /// Precision floor: the minimum representable step is 1 ULP = 2⁻⁶⁴, so /// `FIDELITY_EPS = 1 << 10` (≈ 5.4 × 10⁻¹⁷) is well above noise. #[inline(always)] fn isqrt_u64_fixed(n: u64) -> u64 { if n == 0 { return 0; } // Seed: closest power-of-two approximation. let mut x = 1u64 << ((64 - n.leading_zeros() + 1) / 2); // 8 Newton–Raphson steps for U0F64 convergence. // [IMPROVEMENT 3-B] Two extra steps vs the original 6 to cover fractional // precision; verified by the Kani harness below. macro_rules! newton { () => { x = (x.wrapping_add(n / x)) / 2; }; } newton!(); newton!(); newton!(); newton!(); newton!(); newton!(); newton!(); newton!(); // iterations 7 & 8 x } // ── Kani verification harnesses ────────────────────────────────────────────── #[cfg(kani)] mod verification { use super::*; /// Verify isqrt terminates and does not panic for all u64 inputs. #[kani::proof] #[kani::unwind(8)] fn verify_isqrt() { let n: u64 = kani::any(); let result = isqrt_u64_fixed(n); // Weak post-condition: result² ≤ n (floor property). if n > 0 { let r = result as u128; kani::assert(r * r <= n as u128, "isqrt floor property"); } } /// [IMPROVEMENT 4-B] Prove update_welford never overflows fixed-point /// bounds for the worst-case (maximum) observation. #[kani::proof] #[kani::unwind(1)] fn verify_welford_no_overflow() { let obs_bits: u64 = kani::any(); // [IMPROVEMENT 4-B] Assume maximum legal fixed-point input. kani::assume(obs_bits <= u64::MAX); let obs = U0F64::from_bits(obs_bits); let mut state = FidelityState::new(); // Two updates: covers the delta/delta2 cross-terms. state.update_welford(obs); state.update_welford(U0F64::from_bits(kani::any::<u64>())); // Post-condition: mean and m2 remain in [0, U0F64::MAX]. // saturating_* ops guarantee this, but Kani confirms no panic paths. let _ = state.variance(); let _ = state.std_dev_approx(); } }
